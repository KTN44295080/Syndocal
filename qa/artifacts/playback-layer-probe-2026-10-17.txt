// Temporary measurement probe (not committed). Included at the end of
// `crates/engine/src/lib.rs` `mod tests` to diff the persistence image.

fn probe_diff(path: &str, a: &serde_json::Value, b: &serde_json::Value, out: &mut Vec<String>) {
    use serde_json::Value;
    match (a, b) {
        (Value::Object(x), Value::Object(y)) => {
            let mut keys: Vec<&String> = x.keys().chain(y.keys()).collect();
            keys.sort();
            keys.dedup();
            for k in keys {
                let p = format!("{path}.{k}");
                match (x.get(k), y.get(k)) {
                    (Some(l), Some(r)) => probe_diff(&p, l, r, out),
                    (l, r) => out.push(format!("{p}: {:?} -> {:?}", l, r)),
                }
            }
        }
        (Value::Array(x), Value::Array(y)) if x.len() == y.len() => {
            for (i, (l, r)) in x.iter().zip(y).enumerate() {
                probe_diff(&format!("{path}[{i}]"), l, r, out);
            }
        }
        _ if a == b => {}
        _ => {
            let s = |v: &Value| {
                let t = v.to_string();
                if t.len() > 160 { format!("{}...", &t[..160]) } else { t }
            };
            out.push(format!("{path}: {} -> {}", s(a), s(b)));
        }
    }
}

fn probe_image(runtime: &EngineRuntime) -> serde_json::Value {
    serde_json::to_value(runtime.build_persistence_snapshot()).unwrap()
}

fn probe_report(label: &str, a: &serde_json::Value, b: &serde_json::Value) -> Vec<String> {
    let mut out = Vec::new();
    probe_diff("", a, b, &mut out);
    eprintln!("=== {label}: {} changed path(s)", out.len());
    for line in &out {
        eprintln!("    {line}");
    }
    out
}

fn zz_probe_playback_write_set() {
    // 1. Trigger cue 1 (fixture + video layer + legacy effect target).
    let mut runtime = probe_runtime();
    let authored = probe_image(&runtime);
    eprintln!("base dimmer={:?} pan={:?} cues={:?}", probe_dimmer(&runtime), probe_pan(&runtime),
        runtime.cues.iter().map(|c| (c.id, c.cue_list_id, c.tracking, c.recall_mode)).collect::<Vec<_>>());
    runtime.apply_command(EngineCommand::TriggerCue(1));
    runtime.apply_active_fade(Instant::now() + Duration::from_secs(5));
    let after_a = probe_image(&runtime);
    probe_report("TriggerCue(1) [fixture+video layer+legacy effect]", &authored, &after_a);
    eprintln!("live dimmer={:?} pan={:?}", probe_dimmer(&runtime), probe_pan(&runtime));

    // 2. GO cue 2 (pan only, 1 s fade) in the same list; sample mid-fade.
    runtime.apply_command(EngineCommand::TriggerCue(2));
    let t0 = Instant::now();
    runtime.apply_active_fade(t0 + Duration::from_millis(500));
    let mid = probe_image(&runtime);
    probe_report("TriggerCue(2) mid-fade vs after cue 1", &after_a, &mid);
    eprintln!("mid dimmer={:?} pan={:?} active_fade={}", probe_dimmer(&runtime), probe_pan(&runtime), runtime.active_fade.is_some());

    // 3. Reopen the mid-fade image.
    let mut reopened = child_graph_test_engine();
    reopened
        .load_project_snapshot_checked(serde_json::from_value(mid.clone()).unwrap())
        .unwrap();
    eprintln!(
        "reopened mid: dimmer={:?} pan={:?} active_cue={:?} active_fade={} effect10={:?}",
        probe_dimmer(&reopened), probe_pan(&reopened), reopened.active_cue_id,
        reopened.active_fade.is_some(),
        reopened.effects.iter().map(|e| e.enabled).collect::<Vec<_>>()
    );
    probe_report("reopen(mid) image vs saved mid image", &mid, &probe_image(&reopened));

    // 4. Finish the fade, release cue 2 then cue 1 live; compare to authored.
    runtime.apply_active_fade(t0 + Duration::from_secs(5));
    eprintln!("after cue 2: dimmer={:?} pan={:?} active={:?}", probe_dimmer(&runtime), probe_pan(&runtime), runtime.active_cue_id);
    let after_b = probe_image(&runtime);
    runtime.apply_command(EngineCommand::ReleaseCue(2));
    runtime.apply_active_fade(Instant::now() + Duration::from_secs(5));
    eprintln!("after release 2: dimmer={:?} pan={:?} active={:?} err={:?}", probe_dimmer(&runtime), probe_pan(&runtime), runtime.active_cue_id, runtime.last_error);
    runtime.apply_command(EngineCommand::ReleaseCue(1));
    runtime.apply_active_fade(Instant::now() + Duration::from_secs(5));
    eprintln!("after release 1: dimmer={:?} pan={:?} active={:?} err={:?}", probe_dimmer(&runtime), probe_pan(&runtime), runtime.active_cue_id, runtime.last_error);
    probe_report("live release of both vs authored", &authored, &probe_image(&runtime));

    // 5. Reopen the settled cue-2 image, then release there.
    let mut reopened = child_graph_test_engine();
    reopened
        .load_project_snapshot_checked(serde_json::from_value(after_b.clone()).unwrap())
        .unwrap();
    eprintln!("reopened after_b: dimmer={:?} pan={:?} active={:?}", probe_dimmer(&reopened), probe_pan(&reopened), reopened.active_cue_id);
    probe_report("reopen(after_b) image vs saved", &after_b, &probe_image(&reopened));
    let active = reopened.active_cue_id;
    if let Some(cue) = active {
        reopened.apply_command(EngineCommand::ReleaseCue(cue));
        reopened.apply_active_fade(Instant::now() + Duration::from_secs(5));
        eprintln!("reopened release {cue}: dimmer={:?} pan={:?} active={:?} err={:?}", probe_dimmer(&reopened), probe_pan(&reopened), reopened.active_cue_id, reopened.last_error);
    }
    probe_report("reopened+released vs authored", &authored, &probe_image(&reopened));

    // 6. Fader edit while cue 1 is live, then release cue 1.
    let mut runtime = probe_runtime();
    runtime.apply_command(EngineCommand::TriggerCue(1));
    runtime.apply_active_fade(Instant::now() + Duration::from_secs(5));
    runtime.apply_command(EngineCommand::SetAttribute { fixture_id: 1, attribute: "Dimmer".into(), value: 20_000 });
    eprintln!("fader over live cue: dimmer={:?}", probe_dimmer(&runtime));
    runtime.apply_command(EngineCommand::ReleaseCue(1));
    runtime.apply_active_fade(Instant::now() + Duration::from_secs(5));
    eprintln!("after release: dimmer={:?} (authored base was 1000, fader wrote 20000)", probe_dimmer(&runtime));
}

fn zz_probe_timeline_playback_write_set() {
    let mut runtime = probe_runtime();
    runtime.apply_command(EngineCommand::AddTimelineCueEvent {
        time_beats: None,
        event_id: 50,
        cue_id: 1,
        time_ms: 20,
        track: TimelineTrackKind::Lighting,
        layer_id: None,
    });
    assert_eq!(runtime.last_error, None);
    runtime.apply_command(EngineCommand::AddTimelineAutomation {
        automation_id: 60,
        fixture_id: 1,
        attribute: "Pan".to_string(),
        keyframes: vec![
            AutomationKeyframeSummary { time_ms: 0, value: 0, interpolation: AutomationInterpolation::Linear },
            AutomationKeyframeSummary { time_ms: 1_000, value: 60_000, interpolation: AutomationInterpolation::Linear },
        ],
    });
    eprintln!("automation add err={:?}", runtime.last_error);
    let authored = probe_image(&runtime);
    runtime.last_tick_interval = Duration::from_millis(10);
    runtime.apply_command(EngineCommand::SetTimelinePlaying(true));
    let started = Instant::now();
    let after_play_cmd = probe_image(&runtime);
    probe_report("SetTimelinePlaying(true) alone", &authored, &after_play_cmd);
    for step in 1..=50u64 {
        let now = started + Duration::from_millis(step * 10);
        runtime.advance_timeline(now);
        runtime.advance_pending_cue(now);
        runtime.apply_timeline_automations();
        runtime.apply_active_fade(now);
    }
    eprintln!("timeline pos={} active={:?} dimmer={:?} pan={:?}", runtime.timeline_position_ms, runtime.active_cue_id, probe_dimmer(&runtime), probe_pan(&runtime));
    probe_report("Timeline 500 ms (Scene Block @20 ms + Pan automation)", &authored, &probe_image(&runtime));
    runtime.apply_command(EngineCommand::SetTimelinePlaying(false));
    probe_report("Timeline stopped vs authored", &authored, &probe_image(&runtime));
}
